Integrate ValhallaRaces into your project with Maven
```
<repositories>
    <repository>
        <id>repsy</id>
        <url>https://repo.repsy.io/mvn/athlaeos/valhallammo</url>
    </repository>
</repositories>

<dependencies>
    <dependency>
        <groupId>me.athlaeos</groupId>
        <artifactId>valhallammo-dist</artifactId>
        <version>dev-0.2</version>
    </dependency>
</dependencies>
```
